!!!!!!!!!!!!!!!! 3.) Greatest Common Divisor (GCD) !!!!!!!!!!!!!!

birth {

function_int @gcd(int $a, int $b) {
    int $temp;
    while ($b != 0) {
        $temp = $b;
        $b = $a % $b;
        $a = $temp;
    }
    revive $a;
}

@main_casper() {
    display "Enter first number for GCD: ", "\n";
    str $input1 = input();
    int $num1 = to_int($input1);
    display "Enter second number for GCD: ", "\n";
    str $input2 = input();
    int $num2 = to_int($input2);

    int $result = @gcd($num1, $num2);

    display "The GCD of ", $num1, " and ", $num2, " is: ", $result, "\n";
}

}

!!!!!!!!!!!!!!!! 4.) Least Common Multiple (LCM) - Inline !!!!!!!!!!!!!!

birth {

@main_casper() {
    display "Enter first number for LCM: ", "\n";
    str $input1 = input();
    int $num1 = to_int($input1);
    display "Enter second number for LCM: ", "\n";
    str $input2 = input();
    int $num2 = to_int($input2);

    int $result_lcm = 0;

    check ($num1 == 0 || $num2 == 0) {
        $result_lcm = 0;
    } otherwise {
        int $a = $num1;
        int $b = $num2;
        int $temp;
        while ($b != 0) {
            $temp = $b;
            $b = $a % $b;
            $a = $temp;
        }
        int $gcd_val = $a; 

        check ($gcd_val != 0) {
             $result_lcm = ($num1 * $num2) / $gcd_val;
        } otherwise {
            $result_lcm = 0;
        }
    }

    display "The LCM of ", $num1, " and ", $num2, " is: ", $result_lcm, "\n";
}

!!!!!!!!!!!!!!!! 12.) Simple String Anagram Checker !!!!!!!!!!!!!!

birth {

@main_casper() {
    display "Simple Anagram Checker Program", "\n";
    display "Enter first string: ", "\n";
    str $str1 = input();
    display "Enter second string: ", "\n";
    str $str2 = input();
    
    int $len1 = measure($str1);
    int $len2 = measure($str2);
    
    << Check if lengths are different first (quick check)
    check ($len1 != $len2) {
        display "The strings are not anagrams (different lengths).", "\n";
    } otherwise {
        bln $is_anagram = Day;
        
        << For each position in first string
        int $i = 0;
        while ($i < $len1) {
            str $current_char = $str1[$i];
            
            << Count occurrences in first string
            int $count1 = 0;
            int $j = 0;
            while ($j < $len1) {
                check ($str1[$j] == $current_char) {
                    $count1 = $count1 + 1;
                } otherwise {
                    << Not a match, continue
                }
                $j = $j + 1;
            }
            
            << Count occurrences in second string
            int $count2 = 0;
            $j = 0;
            while ($j < $len2) {
                check ($str2[$j] == $current_char) {
                    $count2 = $count2 + 1;
                } otherwise {
                    << Not a match, continue
                }
                $j = $j + 1;
            }
            
            << If counts don't match, not an anagram
            check ($count1 != $count2) {
                $is_anagram = Night;
                stop; << Exit early
            } otherwise {
                << Counts match for this character, continue
            }
            
            $i = $i + 1;
        }
        
        check ($is_anagram == Day) {
            display "The strings are anagrams!", "\n";
        } otherwise {
            display "The strings are not anagrams.", "\n";
        }
    }
}

}

!!!!!!!!!!!!!!!! 13.) String Reversal !!!!!!!!!!!!!!

birth {

@main_casper() {
    display "String Reversal Program", "\n";
    display "Enter a string to reverse: ", "\n";
    str $input_string = input();
    
    int $length = measure($input_string);
    str $reversed = "";
    
    check ($length == 0) {
        display "You entered an empty string.", "\n";
    } otherwise {
        display "Original string: ", $input_string, "\n";
        
        int $i = $length - 1;
        while ($i >= 0) {
            str $current_char = $input_string[$i];
            $reversed = $reversed + $current_char;
            $i = $i - 1;
        }
        
        display "Reversed string: ", $reversed, "\n";
    }
}

}
