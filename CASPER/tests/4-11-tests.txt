function @decimalToBinary(int $decimalValue) {
    << Check for negative input first
    check ($decimalValue < 0) {
        display "Error: Negative numbers cannot be converted to binary using this method.", "\n";
    } otherwise {
        << Proceed with non-negative conversion
        str $binaryString = "";

        << Handle the edge case of input 0
        check ($decimalValue == 0) {
            $binaryString = "0";
        } otherwise {
            << Loop while the decimal value is greater than 0
            while ($decimalValue > 0) {
                << Get the remainder (0 or 1)
                int $remainder = $decimalValue % 2;
                << Convert remainder to string
                str $remainderStr = to_str($remainder);
                << Prepend the remainder string to the binary string
                $binaryString = $remainderStr + $binaryString; 
                
                << Perform integer division by 2
                << Using temp variable and to_int() to ensure integer result
                int $temp = $decimalValue / 2; 
                $decimalValue = to_int($temp); 
            }
        }

        display "Binary equivalent: ", $binaryString, "\n";
    }
