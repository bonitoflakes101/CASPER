!!!!!!!!!!!!!!!! 1.) PRIME NUMBER !!!!!!!!!!!!!!

birth {

@main_casper() {
    display "Enter a number to check if it's prime: ", "\n";
    int $num = input();
    
    bln $isPrime = Day;
    
    check ($num < 2) {
        $isPrime = Night;
    } otherwise {
        int $i = 2;
        
        while ($i * $i <= $num) {
            check ($num % $i == 0) {
                $isPrime = Night;
                stop;
            }
            $i = $i + 1;
        }
    }
    
    check ($isPrime == Day) {
        display $num, " is a prime number.";
    } otherwise {
        display $num, " is not a prime number.";
    }
}

}
