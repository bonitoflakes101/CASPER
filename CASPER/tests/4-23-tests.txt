!!!!!!!!!!!!!!!!!!! 3.) REVERSE FIBONACCI !!!!!!!!!!!!!!!!!!!!!!!!

birth {

@main_casper() {
    display "Enter the number of Fibonacci numbers to generate: ", "\n";
    int $n = input(); 

    check ($n <= 0) {
        display "Please enter a positive number.", "\n"; 
    } otherwise {
        int $fib_sequence[]; 

        check ($n == 1) {
            $fib_sequence.push(0);
        } otherwise {
            // Start with first Fibonacci number
            $fib_sequence.push(0);
            
            // For n > 1, add second Fibonacci number
            check ($n >= 2) {
                $fib_sequence.push(1);
                
                int $a = 0; 
                int $b = 1;
                int $i = 2; 

                // Generate remaining Fibonacci numbers
                while ($i < $n) {
                    int $next_fib = $a + $b;
                    $fib_sequence.push($next_fib);

                    $a = $b;        
                    $b = $next_fib; 

                    $i = $i + 1;
                }
            } otherwise {
                // This case is logically unreachable (n > 1 but n < 2),
                // but kept to maintain structure
            }
        }

        display "Fibonacci sequence generated. Displaying in reverse order:", "\n";
        // Calculate the actual size of fib_sequence for proper indexing
        int $size = measure($fib_sequence);
        int $j = $size - 1;
        
        while ($j >= 0) {
            display $fib_sequence[$j], " ";
            $j = $j - 1;
        }
        display "\n";
    }
}

}

!!!!!!!!!!!!!!!!!!!!!! 9.) PALINDROME CHECK !!!!!!!!!!!!!!!!

birth{

@main_casper() {
    display "Palindrome Checker", "\n";
    display "Enter a string: ", "\n";
    str $input_str = input();
    int $length = measure($input_str);

    bln $isPalindrome = Day;
    int $left = 0;
    int $right = $length - 1;
    bln $continueCheck = Day; 

    check ($length <= 1) {
        $isPalindrome = Day;
        $continueCheck = Night; 
    } otherwise {
    }

    while ($left < $right && $continueCheck == Day) {
        str $leftChar = $input_str[$left];
        str $rightChar = $input_str[$right];

        check ($leftChar != $rightChar) {
            $isPalindrome = Night; 
            $continueCheck = Night; 
        } otherwise {
            $left = $left + 1;
            $right = $right - 1;
        }
    }

    check ($isPalindrome == Day) {
        display "'", $input_str, "' is a palindrome.", "\n";
    } otherwise {
        display "'", $input_str, "' is not a palindrome.", "\n";
    }
}

}
