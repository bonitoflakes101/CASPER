    check ($n <= 0) {
        display "Please enter a positive number.", "\n"; 
    } otherwise {
        int $fib_sequence[]; 

        check ($n == 1) {
            $fib_sequence.push(0);
        } otherwise {

            $fib_sequence.push(0);

            check ($n >= 2) {
                $fib_sequence.push(1);
                
                int $a = 0; 
                int $b = 1;
                int $i = 2; 

                while ($i < $n) {
                    int $next_fib = $a + $b;
                    $fib_sequence.push($next_fib);

                    $a = $b;        
                    $b = $next_fib; 

                    $i = $i + 1;
                }
            } otherwise {

            }
        }

        display "Fibonacci sequence generated. Displaying in reverse order:", "\n";

        int $size = measure($fib_sequence);
        int $j = $size - 1;
        
        while ($j >= 0) {
            display $fib_sequence[$j], " ";
            $j = $j - 1;
        }
        display "\n";
    }

!!!!!!!!!!!!!!! 4.) PERMUTATION OF A GIVEN STRING !!!!!!!!!!!!!!!!!!!!


birth {

<< Helper function to swap characters at positions i and j in a string
function_str @swap(str $input, int $i, int $j) {
    str $chars[];
    int $len = measure($input);
    int $k = 0;
    while ($k < $len) {
        str $char = $input[$k];
        $chars.push($char);
        $k = $k + 1;
    }
    str $temp = $chars[$i];
    $chars[$i] = $chars[$j];
    $chars[$j] = $temp;
    str $result = "";
    $k = 0;
    while ($k < $len) {
        $result = $result + $chars[$k];
        $k = $k + 1;
    }
    revive $result;
}

<< Recursive function to generate permutations
function @findPermutations(str $currentString, int $start, int $end) {

    check ($start == $end) {
        display $currentString, "\n";
    } otherwise {
        int $i = $start;
        while ($i <= $end) {
            str $swapped = @swap($currentString, $start, $i);
            int $nextPos = $start + 1;
            @findPermutations($swapped, $nextPos, $end);
            $i = $i + 1;
        }
    }
}


@main_casper() {
    display "Permutation Generator", "\n";
    display "Enter a string: ", "\n";
    str $inputString = input();
    int $length = measure($inputString);
    int $lastIndex = $length - 1;
    @findPermutations($inputString, 0, $lastIndex);
}

birth{

@main_casper() {
    display "Palindrome Checker", "\n";
    display "Enter a string: ", "\n";
    str $input_str = input();
    int $length = measure($input_str);

    bln $isPalindrome = Day;
    int $left = 0;
    int $right = $length - 1;
    bln $continueCheck = Day; 

    check ($length <= 1) {
        $isPalindrome = Day;
        $continueCheck = Night; 
    } otherwise {
    }

    while ($left < $right && $continueCheck == Day) {
        str $leftChar = $input_str[$left];
        str $rightChar = $input_str[$right];

        check ($leftChar != $rightChar) {
            $isPalindrome = Night; 
            $continueCheck = Night; 
        } otherwise {
            $left = $left + 1;
            $right = $right - 1;
        }
    }

    check ($isPalindrome == Day) {
        display "'", $input_str, "' is a palindrome.", "\n";
    } otherwise {
        display "'", $input_str, "' is not a palindrome.", "\n";
    }
}

}
