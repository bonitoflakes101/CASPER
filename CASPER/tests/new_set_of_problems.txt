!!!!!!!!!!!!!!!! 2.) Average of List Elements !!!!!!!!!!!!!!

birth {

@main_casper() {
    << Define a sample list directly
    flt $numbers[] = [10.0, 20.5, 5.0, 15.5, 30.0];
    flt $sum = 0.0;
    int $size = measure($numbers);

    display "List: [10.0, 20.5, 5.0, 15.5, 30.0]", "\n";

    check ($size == 0) {
        display "Cannot calculate average of an empty list.", "\n";
    } otherwise {
        int $i = 0;
        while ($i < $size) {
            $sum = $sum + $numbers[$i];
            $i = $i + 1;
        }
        flt $flt_size = to_flt($size);
        flt $average = $sum / $flt_size; 

        display "The average of the list elements is: ", $average, "\n";
    }
}

    << Test case 2 (unsorted): int $numbers[] = [2, 8, 5, 12, 15];

    display "List: [2, 5, 8, 12, 15]", "\\n"; << Adjust if changing list

    int $size = measure($numbers);

    check ($size > 1) {
        int $i = 0;
        while ($i < $size - 1) {
            check ($numbers[$i] > $numbers[$i+1]) {
                $isSorted = Night;

                << Elements are in order so far
            }
            $i = $i + 1;
        }

        << A list with 0 or 1 element is considered sorted
        $isSorted = Day;
    }

    check ($isSorted == Day) {
        display "The list is sorted in ascending order.", "\\n";
    } otherwise {
        display "The list is NOT sorted in ascending order.", "\\n";

!!!!!!!!!!!!!!!! 7.) Exponentiation (Base^Exponent) using Loop !!!!!!!!!!!!!!

birth {

@main_casper() {
    display "Exponentiation Calculator (base^exponent)", "\\n";
    display "Enter base: ", "\\n";
    str $base_str = input();
    int $base = to_int($base_str);

    display "Enter non-negative integer exponent: ", "\\n";
    str $exp_str = input();
    int $exponent = to_int($exp_str);

    int $result = 1;

    check ($exponent < 0) {
        display "Exponent must be non-negative for this simple calculator.", "\\n";
    } otherwise_check ($exponent == 0) {
        $result = 1; << Any base to power 0 is 1
        display $base, " ^ ", $exponent, " = ", $result, "\\n";
    } otherwise {
        int $i = 0;
        while ($i < $exponent) {
            $result = $result * $base;
            $i = $i + 1;
        }
        display $base, " ^ ", $exponent, " = ", $result, "\\n";
    }
}

}

!!!!!!!!!!!!!!!! 8.) Triangle Type (Equilateral, Isosceles, Scalene) !!!!!!!!!!!!!!

birth {

@main_casper() {
    display "Triangle Type Identifier", "\\n";
    display "Enter length of side 1: ", "\\n";
    str $s1_str = input();
    flt $s1 = to_flt($s1_str);

    display "Enter length of side 2: ", "\\n";
    str $s2_str = input();
    flt $s2 = to_flt($s2_str);

    display "Enter length of side 3: ", "\\n";
    str $s3_str = input();
    flt $s3 = to_flt($s3_str);

    << Basic check for valid triangle (sum of two sides > third side)
    bln $valid_triangle = Day;
    check ($s1 + $s2 <= $s3 || $s1 + $s3 <= $s2 || $s2 + $s3 <= $s1 || $s1 <= 0 || $s2 <= 0 || $s3 <= 0) {
        $valid_triangle = Night;
    } otherwise {}

    check ($valid_triangle == Day) {
        check ($s1 == $s2 && $s2 == $s3) {
            display "The triangle is Equilateral.", "\\n";
        } otherwise_check ($s1 == $s2 || $s1 == $s3 || $s2 == $s3) {
            display "The triangle is Isosceles.", "\\n";
        } otherwise {
            display "The triangle is Scalene.", "\\n";
        }
    } otherwise {
        display "The given side lengths do not form a valid triangle.", "\\n";
    }
}

}

    display "Enter a positive integer greater than 1: ", "\\n";
    str $n_str = input();
    int $n = to_int($n_str);

    check ($n <= 1) {
        display "Please enter an integer greater than 1.", "\\n";
    } otherwise {
        display "Sequence: ", $n;
        while ($n != 1) {
            check ($n % 2 == 0) {
                << Even number
                $n = to_int($n / 2);
            } otherwise {
                << Odd number
                $n = 3 * $n + 1;
            }

        }

    }
}

!!!!!!!!!!!!!!!! 12.) Binary Search (Sorted List) !!!!!!!!!!!!!!
<< Searches for a target value in a pre-sorted list of integers. >>

birth {

@main_casper() {
    << Define a pre-sorted list >>
    int $sorted_list[] = [2, 5, 8, 12, 16, 23, 38, 56, 72, 91];
    int $size = measure($sorted_list);

    display "Sorted List: [2, 5, 8, 12, 16, 23, 38, 56, 72, 91]", "\n";
    display "Enter the number to search for: ", "\n";
    str $target_str = input();
    int $target = to_int($target_str);

    int $low = 0;
    int $high = $size - 1;
    bln $found = Night;
    int $mid = 0;
    int $mid_val = 0;

    while ($low <= $high) {
        int $range = $high - $low;
        int $range_int = to_int($range / 2);
        $mid = $low + $range_int;
        $mid_val = $sorted_list[$mid];

        check ($mid_val == $target) {
            $found = Day;
            stop; << Exit loop, target found >>
        } otherwise_check ($mid_val < $target) {
            $low = $mid + 1; 
        } otherwise {
            $high = $mid - 1; 
        }
    }

    check ($found == Day) {
        display $target, " was found in the list at index ", $mid, ".", "\n";
    } otherwise {
        display $target, " was not found in the list.", "\n";
    }
}

}

!!!!!!!!!!!!!!!! 14.) Find Mode in a List (Simple Integer List) !!!!!!!!!!!!!!
<< Finds the number that appears most frequently in a hardcoded list. >>
<< Assumes there is one unique mode for simplicity. >>

birth {

@main_casper() {
    int $numbers[] = [2, 3, 5, 3, 7, 8, 5, 3, 9, 3];
    int $size = measure($numbers);

    display "List: [2, 3, 5, 3, 7, 8, 5, 3, 9, 3]", "\n";

    check ($size == 0) {
        display "Cannot find mode of an empty list.", "\n";
    } otherwise {
        int $mode = $numbers[0];
        int $max_count = 0;
        int $i = 0;

        << Iterate through each unique potential mode >>
        while ($i < $size) {
            int $current_number = $numbers[$i];
            int $current_count = 0;
            int $j = 0;

            << Count occurrences of current_number >>
            while ($j < $size) {
                check ($numbers[$j] == $current_number) {
                    $current_count = $current_count + 1;
                } otherwise { }
                $j = $j + 1;
            }

            << Check if this number is more frequent than the current mode >>
            check ($current_count > $max_count) {
                $max_count = $current_count;
                $mode = $current_number;
            } otherwise { }

            $i = $i + 1;
        }

        display "The mode of the list is: ", $mode, " (appeared ", $max_count, " times)", "\\";
    }
}

}

!!!!!!!!!!!!!!!! 15.) Sum of Even/Odd Numbers in a List !!!!!!!!!!!!!!

!!!!!!!!!!!!!!!! 16.) Caesar Cipher (Simple Shift +3, Uppercase) !!!!!!!!!!!!!!
<< Encrypts a hardcoded uppercase string using a Caesar cipher with a shift of 3. >>

birth {

@main_casper() {
    str $plaintext = "HELLO";
    str $alphabet = "ABCDEFGHIJKLMNOPQRSTUVWXYZ";
    int $shift = 3;
    int $alphabet_len = measure($alphabet);
    str $ciphertext = "";
    int $text_len = measure($plaintext);

    display "Plaintext: ", $plaintext, "\n";
    display "Shift: ", $shift, "\n";

    int $i = 0;
    while ($i < $text_len) {
        str $char_to_encrypt = $plaintext[$i];
        str $encrypted_char = "?"; << Default if not found >>
        int $found_index = -1;

        << Find the index of the character in the alphabet >>
        int $j = 0;
        while ($j < $alphabet_len) {
            check ($alphabet[$j] == $char_to_encrypt) {
                $found_index = $j;
                stop;
            } otherwise {}
            $j = $j + 1;
        }

        << If found, calculate the shifted index and get the new char >>
        check ($found_index != -1) {
            int $shifted_index = ($found_index + $shift) % $alphabet_len;
            $encrypted_char = $alphabet[$shifted_index];
        } otherwise {
           << Character not in alphabet, keep default '?' >>
        }

        $ciphertext = $ciphertext + $encrypted_char;
        $i = $i + 1;
    }

    display "Ciphertext: ", $ciphertext, "\n";
}

}
